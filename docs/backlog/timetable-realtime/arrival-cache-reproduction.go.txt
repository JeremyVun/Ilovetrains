package api

import (
	"encoding/json"
	"testing"
	"time"

	"trains/internal/tfnsw"
)

// Temporary diagnostic: verifies the observed cache failure, not desired behavior.
func TestInvestigateArrivalChangeDuringPastWindow(t *testing.T) {
	upstream := &fakeUpstream{departures: sampleDepartures()}
	server, handler := pinnedServer(t, upstream)
	clock := testNow
	server.now = func() time.Time { return clock }
	server.departures.SetClock(func() time.Time { return clock })
	server.departuresPast.SetClock(func() time.Time { return clock })
	initial := testNow.Add(5 * time.Minute).Format(time.RFC3339)
	delayed := testNow.Add(10 * time.Minute).Format(time.RFC3339)
	upstream.departures.GeneratedAt = testNow.Format(time.RFC3339)
	upstream.departures.Journeys[0].Departure.Scheduled = testNow.Add(-25 * time.Minute).Format(time.RFC3339)
	upstream.departures.Journeys[0].Arrival = tfnsw.Arrival{Scheduled: initial, Estimated: &initial}
	target := departuresAt(-25 * time.Minute)
	first := get(t, handler, target)
	// Replace the upstream response without mutating the cached slice.
	newResponse := *upstream.departures
	newResponse.Journeys = append([]tfnsw.Journey(nil), upstream.departures.Journeys...)
	newResponse.Journeys[0].Arrival.Estimated = &delayed
	upstream.departures = &newResponse
	clock = clock.Add(3 * time.Minute)
	second := get(t, handler, target)
	var response tfnsw.DeparturesResponse
	if err := json.Unmarshal(second.Body.Bytes(), &response); err != nil { t.Fatal(err) }
	if upstream.departureCalls.Load() != 1 || response.Journeys[0].Arrival.Estimated == nil || *response.Journeys[0].Arrival.Estimated != initial {
		t.Fatal("expected current implementation to reproduce a cached, unchanged arrival")
	}
	t.Logf("still in transit: now=%s, shown arrival=%s, upstream arrival=%s, upstream fetches=%d, cache=%s", clock.Format(time.RFC3339), *response.Journeys[0].Arrival.Estimated, delayed, upstream.departureCalls.Load(), first.Header().Get("Cache-Control"))
}
